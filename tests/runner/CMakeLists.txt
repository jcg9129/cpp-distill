if(TARGET distill)
  add_dependencies(distill_pipeline_cases distill_prune_worker)

  list(LENGTH CAIDE_EXPECTATIONS CAIDE_COUNT)
  if(NOT CAIDE_COUNT EQUAL 81)
    message(FATAL_ERROR "The caide port requires exactly 81 case expectations")
  endif()
  foreach(EXPECTATION IN LISTS CAIDE_EXPECTATIONS)
    get_filename_component(CASE_DIR "${EXPECTATION}" DIRECTORY)
    get_filename_component(CASE_NAME "${CASE_DIR}" NAME)
    add_test(NAME caide_${CASE_NAME} COMMAND distill_pipeline_cases --caide $<TARGET_FILE:distill_prune_worker> "${CASE_DIR}")
    set_tests_properties(caide_${CASE_NAME} PROPERTIES TIMEOUT 120 LABELS caide)
  endforeach()
  add_test(NAME pipeline_cli COMMAND distill_pipeline_cases --cli $<TARGET_FILE:distill>)
  set_tests_properties(pipeline_cli PROPERTIES TIMEOUT 120)
  add_test(NAME pipeline_pruning COMMAND distill_pipeline_cases --worker $<TARGET_FILE:distill_prune_worker>)
  set_tests_properties(pipeline_pruning PROPERTIES TIMEOUT 120)
  add_test(NAME pipeline_pruning_d3c COMMAND distill_pipeline_cases --worker $<TARGET_FILE:distill_prune_worker> --d3c)
  set_tests_properties(pipeline_pruning_d3c PROPERTIES TIMEOUT 120)
  add_test(NAME pipeline_cli_d3c COMMAND distill_pipeline_cases --cli $<TARGET_FILE:distill> --d3c)
  set_tests_properties(pipeline_cli_d3c PROPERTIES TIMEOUT 120)
  add_test(NAME pipeline_pruning_d3d COMMAND distill_pipeline_cases --worker $<TARGET_FILE:distill_prune_worker> --d3d)
  set_tests_properties(pipeline_pruning_d3d PROPERTIES TIMEOUT 120)
  add_test(NAME pipeline_cli_d3d COMMAND distill_pipeline_cases --cli $<TARGET_FILE:distill> --d3d)
  set_tests_properties(pipeline_cli_d3d PROPERTIES TIMEOUT 120)
  add_test(NAME pipeline_clang COMMAND distill_pipeline_cases --worker $<TARGET_FILE:distill_prune_worker>)
  set_tests_properties(pipeline_clang PROPERTIES TIMEOUT 120 ENVIRONMENT "DISTILL_TEST_EVALUATOR=clang")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME cli_options COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/cli_options.py $<TARGET_FILE:distill>)
  set_tests_properties(cli_options PROPERTIES TIMEOUT 120)
  add_test(NAME cli_help COMMAND distill --help)
  set_tests_properties(cli_help PROPERTIES TIMEOUT 120 PASS_REGULAR_EXPRESSION "Usage: distill")
endif()
