add_executable(distill_inline_cases inline_cases.cpp)
target_link_libraries(distill_inline_cases PRIVATE distill_core)
target_compile_definitions(distill_inline_cases PRIVATE DISTILL_INLINE_CASES="${PROJECT_SOURCE_DIR}/tests/cases/inline")
add_test(NAME inline_cases COMMAND distill_inline_cases)
set_tests_properties(inline_cases PROPERTIES TIMEOUT 120)

add_executable(distill_resolve_cases resolve_cases.cpp)
target_link_libraries(distill_resolve_cases PRIVATE distill_core distill_test_evaluate)
target_compile_definitions(distill_resolve_cases PRIVATE DISTILL_RESOLVE_CASES="${PROJECT_SOURCE_DIR}/tests/cases/resolve")
add_test(NAME resolve_cases COMMAND distill_resolve_cases)
set_tests_properties(resolve_cases PROPERTIES TIMEOUT 120)
if(TARGET distill_clang)
  add_test(NAME resolve_cases_clang COMMAND distill_resolve_cases)
  set_tests_properties(resolve_cases_clang PROPERTIES TIMEOUT 120 ENVIRONMENT "DISTILL_TEST_EVALUATOR=clang")
  add_test(NAME resolve_cases_differential COMMAND distill_resolve_cases)
  set_tests_properties(resolve_cases_differential PROPERTIES TIMEOUT 120 ENVIRONMENT "DISTILL_TEST_EVALUATOR=differential")
endif()

add_executable(distill_pipeline_cases pipeline_cases.cpp)
target_link_libraries(distill_pipeline_cases PRIVATE distill_core distill_test_evaluate)
target_compile_definitions(distill_pipeline_cases PRIVATE DISTILL_PIPELINE_CASES="${PROJECT_SOURCE_DIR}/tests/cases/pipeline")
add_test(NAME pipeline_cases COMMAND distill_pipeline_cases)
set_tests_properties(pipeline_cases PROPERTIES TIMEOUT 120)
if(TARGET distill)
  add_test(NAME pipeline_cli COMMAND distill_pipeline_cases --cli $<TARGET_FILE:distill>)
  set_tests_properties(pipeline_cli PROPERTIES TIMEOUT 120)
  add_test(NAME pipeline_clang COMMAND distill_pipeline_cases)
  set_tests_properties(pipeline_clang PROPERTIES TIMEOUT 120 ENVIRONMENT "DISTILL_TEST_EVALUATOR=clang")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME cli_options COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/cli_options.py $<TARGET_FILE:distill>)
  set_tests_properties(cli_options PROPERTIES TIMEOUT 120)
  add_test(NAME cli_help COMMAND distill --help)
  set_tests_properties(cli_help PROPERTIES TIMEOUT 120 PASS_REGULAR_EXPRESSION "Usage: distill")
endif()
